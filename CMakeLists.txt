option(ANYVOC_BUILD_MIDI_ALLOCATION_PROBE "Build isolated, instrumented MIDI allocation diagnostics (never distribute)" OFF)
set(allocationMarker "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/ANYVOC_MIDI_ALLOCATION_DIAGNOSTIC.json")
if(EXISTS "${allocationMarker}" AND NOT ANYVOC_BUILD_MIDI_ALLOCATION_PROBE)
    message(FATAL_ERROR "An isolated allocation snapshot may only build its opt-in diagnostic target.")
endif()
if(ANYVOC_BUILD_MIDI_ALLOCATION_PROBE)
    if(NOT EXISTS "${allocationMarker}")
        message(FATAL_ERROR "Use scripts/prepare-midi-allocation-probe.py to create a new isolated source tree first. Never instrument the release JUCE checkout.")
    endif()
    file(READ "${allocationMarker}" allocationManifest)
    string(JSON allocationSchema GET "${allocationManifest}" schema)
    string(JSON allocationDiagnostic GET "${allocationManifest}" diagnostic_only)
    string(JSON allocationJuce GET "${allocationManifest}" juce_commit)
    string(JSON allocationRecorderHash GET "${allocationManifest}" recorder_sha256)
    string(JSON allocationPatchCount LENGTH "${allocationManifest}" patches)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Tests/MidiAllocationRecorder.h" allocationActualRecorderHash)
    if(NOT allocationSchema EQUAL 1 OR NOT allocationDiagnostic OR
       NOT allocationJuce STREQUAL "29396c22c93392d6738e021b83196283d6e4d850" OR
       NOT allocationRecorderHash STREQUAL allocationActualRecorderHash OR NOT allocationPatchCount EQUAL 3)
        message(FATAL_ERROR "Invalid or changed isolated allocation diagnostic snapshot.")
    endif()
    foreach(allocationIndex RANGE 0 2)
        string(JSON allocationPath GET "${allocationManifest}" patches ${allocationIndex} path)
        string(JSON allocationHash GET "${allocationManifest}" patches ${allocationIndex} after_sha256)
        file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${allocationPath}" allocationActualHash)
        if(NOT allocationHash STREQUAL allocationActualHash)
            message(FATAL_ERROR "Changed isolated JUCE allocation hook: ${allocationPath}")
        endif()
    endforeach()
endif()

# Instrumented binaries are diagnostic artifacts, not plugin distribution files.
# Both targets force-include the same inline TLS recorder, including JUCE modules.
if(ANYVOC_BUILD_MIDI_ALLOCATION_PROBE)
    if(ANYVOC_BUILD_TESTS OR ANYVOC_BUILD_REALTIME_PROBE)
        message(FATAL_ERROR "Configure isolated allocation diagnostics with ANYVOC_BUILD_TESTS=OFF and ANYVOC_BUILD_REALTIME_PROBE=OFF.")
    endif()
    juce_add_console_app(AnyvocMidiAllocationProbe PRODUCT_NAME "Anyvoc MIDI Allocation Probe")
    juce_generate_juce_header(AnyvocMidiAllocationProbe)
    target_sources(AnyvocMidiAllocationProbe PRIVATE Tests/MidiAllocationProbe.cpp)
    target_include_directories(AnyvocMidiAllocationProbe PRIVATE Source Tests)
    target_compile_definitions(AnyvocMidiAllocationProbe PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
    target_link_libraries(AnyvocMidiAllocationProbe PRIVATE Anyvoc juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography)
    foreach(allocationTarget IN ITEMS Anyvoc AnyvocMidiAllocationProbe)
        target_compile_definitions(${allocationTarget} PRIVATE ANYVOC_MIDI_ALLOCATION_DIAGNOSTIC=1)
        if(MSVC)
            target_compile_options(${allocationTarget} PRIVATE /utf-8 "/FI${CMAKE_CURRENT_SOURCE_DIR}/Tests/MidiAllocationRecorder.h")
        else()
            target_compile_options(${allocationTarget} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/Tests/MidiAllocationRecorder.h")
        endif()
    endforeach()
    set_target_properties(Anyvoc_Standalone Anyvoc_VST3 PROPERTIES EXCLUDE_FROM_ALL TRUE)
    message(STATUS "MIDI allocation diagnostic: isolated source only; build the AnyvocMidiAllocationProbe target, never package these binaries.")
endif()

# JUCE 8.0.12 omits Info.txt from its Windows resource-generation dependencies.
# Without this edge, an incremental version bump can retain the old FileVersion.
if(WIN32)
    foreach(resourceTarget IN ITEMS Anyvoc AnyvocEngineTests AnyvocProcessorTests AnyvocWavetableTests AnyvocPresetLibraryTests AnyvocRealtimeProbe AnyvocMidiAllocationProbe)
        if(TARGET ${resourceTarget}_rc_lib)
            get_target_property(resourceDirectory ${resourceTarget} JUCE_GENERATED_SOURCES_DIRECTORY)
            get_target_property(resourceInfo ${resourceTarget} JUCE_INFO_FILE)
            add_custom_command(OUTPUT "${resourceDirectory}/${resourceTarget}_resources.rc"
                APPEND DEPENDS "${resourceInfo}")
        endif()
    endforeach()
endif()
